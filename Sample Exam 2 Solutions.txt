//Problem 2 : Solution

 public static void rightTriangle (int n){
	if(n == 0){
		return;
	}

	if (n > 0){
		for(int i = 0; i < n; i++){
			System.out.print("*");

		}
		System.out.println();
	}
	rightTriangle(n-1);


}

//Problem 1: Solution

public static void rightTriangle( int n ){
	if(n == 0){
		return;
	}

	if (n > 0){
		for (int i = n; i > 0; i--){
			System.out.print("*")

		}
		System.out.println();
	}
	rightTriangle(n-1);


}
